cmake_minimum_required(VERSION 3.20)
project(ats_gateway_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
option(ENABLE_SANITIZERS "Address and undefined behavior sanitizers" ON)
if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include(../components/bacnet_stack/bacnet_sources.cmake)
add_library(bacnet_stack STATIC ${BACNET_SOURCES})
target_include_directories(bacnet_stack PUBLIC ${BACNET_SRC} ../components/bacnet_stack/include)
target_compile_definitions(bacnet_stack PUBLIC ${BACNET_DEFINITIONS})
target_compile_options(bacnet_stack PRIVATE -Wno-format -Wno-unused-function)
add_library(ats_model STATIC ../main/ats_model.c)
target_include_directories(ats_model PUBLIC ../main)
target_compile_options(ats_model PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ats_model PUBLIC m)
add_library(gateway_core STATIC ../main/gateway_bacnet.c ../main/bip_port.c)
target_include_directories(gateway_core PUBLIC ../main)
target_link_libraries(gateway_core PUBLIC ats_model bacnet_stack)
target_compile_options(gateway_core PRIVATE -Wall -Wextra -Werror)
add_library(modbus STATIC ../main/modbus_tcp.c)
target_include_directories(modbus PUBLIC ../main)
target_compile_options(modbus PRIVATE -Wall -Wextra -Werror)
add_library(cjson STATIC ../third_party/cJSON/cJSON.c)
target_include_directories(cjson PUBLIC ../third_party/cJSON)
add_library(custom_map STATIC ../main/custom_map.c)
target_include_directories(custom_map PUBLIC ../main)
target_link_libraries(custom_map PUBLIC modbus m)
target_compile_options(custom_map PRIVATE -Wall -Wextra -Werror)
add_library(gateway_config STATIC ../main/gateway_config.c)
target_link_libraries(gateway_config PUBLIC custom_map ats_model cjson)
target_compile_options(gateway_config PRIVATE -Wall -Wextra -Werror)
add_executable(test_custom_map test_custom_map.c ../main/custom_map.c)
target_include_directories(test_custom_map PRIVATE ../main)
target_link_libraries(test_custom_map PRIVATE m)
target_compile_options(test_custom_map PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME custom_map COMMAND test_custom_map)
add_executable(test_gateway_config test_gateway_config.c)
target_link_libraries(test_gateway_config PRIVATE gateway_config)
target_compile_options(test_gateway_config PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME gateway_config COMMAND test_gateway_config ${CMAKE_CURRENT_SOURCE_DIR}/../main/web/template.csv)
add_executable(test_gateway_web test_gateway_web.c ../main/gateway_web.c)
target_include_directories(test_gateway_web BEFORE PRIVATE web_stubs)
target_link_libraries(test_gateway_web PRIVATE gateway_config)
target_compile_options(test_gateway_web PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME gateway_web COMMAND test_gateway_web)
add_executable(test_custom_integration test_custom_integration.c)
target_link_libraries(test_custom_integration PRIVATE gateway_core custom_map Threads::Threads)
target_compile_options(test_custom_integration PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME custom_integration COMMAND test_custom_integration)
foreach(name ats_model gateway_poll modbus_tcp gateway_bacnet)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_${name}.c)
        add_executable(test_${name} test_${name}.c)
        target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        if(name STREQUAL "gateway_poll")
            target_sources(test_${name} PRIVATE ../main/gateway_poll.c)
            target_link_libraries(test_${name} PRIVATE ats_model)
        elseif(name STREQUAL "ats_model")
            target_link_libraries(test_${name} PRIVATE ats_model)
        elseif(name STREQUAL "modbus_tcp")
            target_link_libraries(test_${name} PRIVATE modbus Threads::Threads)
        else()
            target_link_libraries(test_${name} PRIVATE gateway_core)
        endif()
        add_test(NAME ${name} COMMAND test_${name})
    endif()
endforeach()
add_executable(native_gateway ../tools/native_gateway.c ../main/gateway_poll.c)
target_link_libraries(native_gateway PRIVATE gateway_core modbus Threads::Threads)
target_compile_options(native_gateway PRIVATE -Wall -Wextra -Werror)
