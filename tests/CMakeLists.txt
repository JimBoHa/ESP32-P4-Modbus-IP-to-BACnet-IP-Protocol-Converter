cmake_minimum_required(VERSION 3.20)
project(ats_gateway_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
option(ENABLE_SANITIZERS "Address and undefined behavior sanitizers" ON)
if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include(../components/bacnet_stack/bacnet_sources.cmake)
add_library(bacnet_stack STATIC ${BACNET_SOURCES})
target_include_directories(bacnet_stack PUBLIC ${BACNET_SRC} ../components/bacnet_stack/include)
target_compile_definitions(bacnet_stack PUBLIC ${BACNET_DEFINITIONS})
target_compile_options(bacnet_stack PRIVATE -Wno-format -Wno-unused-function)
add_library(ats_model STATIC ../main/ats_model.c)
target_include_directories(ats_model PUBLIC ../main)
target_compile_options(ats_model PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ats_model PUBLIC m)
add_library(gateway_core STATIC ../main/gateway_bacnet.c ../main/bip_port.c)
target_include_directories(gateway_core PUBLIC ../main)
target_link_libraries(gateway_core PUBLIC ats_model bacnet_stack)
target_compile_options(gateway_core PRIVATE -Wall -Wextra -Werror)
add_library(modbus STATIC ../main/modbus_tcp.c)
target_include_directories(modbus PUBLIC ../main)
target_compile_options(modbus PRIVATE -Wall -Wextra -Werror)
foreach(name ats_model gateway_poll modbus_tcp gateway_bacnet)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/test_${name}.c)
        add_executable(test_${name} test_${name}.c)
        target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        if(name STREQUAL "gateway_poll")
            target_sources(test_${name} PRIVATE ../main/gateway_poll.c)
            target_link_libraries(test_${name} PRIVATE ats_model)
        elseif(name STREQUAL "ats_model")
            target_link_libraries(test_${name} PRIVATE ats_model)
        elseif(name STREQUAL "modbus_tcp")
            target_link_libraries(test_${name} PRIVATE modbus Threads::Threads)
        else()
            target_link_libraries(test_${name} PRIVATE gateway_core)
        endif()
        add_test(NAME ${name} COMMAND test_${name})
    endif()
endforeach()
add_executable(native_gateway ../tools/native_gateway.c ../main/gateway_poll.c)
target_link_libraries(native_gateway PRIVATE gateway_core modbus Threads::Threads)
target_compile_options(native_gateway PRIVATE -Wall -Wextra -Werror)
